# Catchy Catcher processing server dependencies.
#
# 1. Install the CUDA build of PyTorch first (target: NVIDIA RTX 4060):
#
#      pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121
#
#    On a CPU-only machine the same packages install from PyPI, but processing
#    will be dramatically slower.
#
# 2. Then install everything below:
#
#      pip install -r requirements.txt
#
# `ffmpeg` must also be on PATH (see README.md).

# Stem separation
demucs

# Transcription + forced alignment
whisperx

# Tempo detection
librosa

# Server
fastapi
uvicorn
# WebSocket support for the shared-microphone relay.
websockets
python-multipart

# Share tunnel for the web frontend. Pinned below 6 on purpose: gradio 6
# requires huggingface-hub>=2, which breaks whisperx/transformers.
gradio<6

# Numerics (torch also depends on numpy, but the pipeline imports it directly)
numpy

# Phonetics for rhyme detection in line grouping. The CMU pronunciation
# dictionary; if it is missing the pipeline falls back to a spelling rule
# instead of failing.
pronouncing
